cmake_minimum_required(VERSION 3.20)
project(KtgaConsole VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This native client targets Windows 10/11; the server remains platform-independent.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz)
FetchContent_MakeAvailable(json)
add_executable(KtgaConsole WIN32 src/main.cpp src/api.cpp src/model.cpp src/session.cpp)
target_compile_definitions(KtgaConsole PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(KtgaConsole PRIVATE nlohmann_json::nlohmann_json winhttp gdi32 user32 comctl32 comdlg32 shell32 dwmapi ole32 advapi32 uxtheme)
if(MSVC)
  target_compile_options(KtgaConsole PRIVATE /utf-8 /W4 /EHsc)
else()
  target_compile_options(KtgaConsole PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(KtgaConsole PRIVATE -municode -static)
endif()
